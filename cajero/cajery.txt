int esLetra(char c)
{
    return isalpha(c);
}

enum TipoCaracter
{
    ES_LETRA,
    ES_NUMERO,
    NO_ES_LETRA_NI_NUMERO
};

enum TipoCaracter validarCaracter(char caracter)
{
    if (isalpha(caracter) != 0)
    {
        return ES_LETRA;
    }
    else if (isdigit(caracter) != 0)
    {
        return ES_NUMERO;
    }
    else
    {
        return NO_ES_LETRA_NI_NUMERO;
    }
}

/*Función para realizar operaciones en el Cajero ATM - UNI*/
void cajeroATM()
{
    float saldo = 0;
    int opcion;
    float cantidad;
    clrscr();
    do
    {
        printf("\n\nCajero ATM - UNI\n");
        printf("1. Ingresar Saldo en cuenta\n");
        printf("2. Retirar saldo de cuenta\n");
        printf("3. Consultar saldo de cuenta\n");
        printf("4. Volver al Menu Principal\n");
        printf("Ingrese su opcion: ");
        scanf("%d", &opcion);
        switch (opcion)
        {
        case 1:
            clrscr();
            printf("Ingrese la cantidad a ingresar: ");
            scanf("%f", &cantidad);

            if (cantidad < 0 || cantidad > 10000)
            {
                printf("Cantidad no valida. Debe ser mayor que 0 y menor o igual a 10,000.\n");
            }
            else
            {
                saldo += cantidad;
                printf("Saldo actual: %.2f\n", saldo);
            }
            getch();
            clrscr();
            break;
        case 2:
            clrscr();
            printf("Ingrese la cantidad a retirar: ");
            scanf("%f", &cantidad);
            if (cantidad < 0 || cantidad > saldo)
            {
                printf("Cantidad no valida. Debe ser mayor que 0 y menor o igual al saldo actual.\n");
            }
            else
            {
                saldo -= cantidad;
                printf("Saldo actual: %.2f\n", saldo);
            }
            getch();
            clrscr();
            break;
        case 3:
            clrscr();
            printf("Saldo actual: %.2f\n", saldo);
            getch();
            clrscr();
            break;
        case 4:
            menu();
            break;
        default:
            printf("Opcion no valida. Por favor, ingrese una opcion válida.\n");
            getch();
            clrscr();
        }
    } while (opcion != 4);
}

/*Funcion que me da el cuadro de los titulos*/
void recuadro(int x, int y, int xi, int yi)
{
    int z;
    for (z = x; z <= xi; z++)
    {
        gotoxy(z, y);
        printf("%c", 196);
        gotoxy(z, yi);
        printf("%c", 196);
    }

    for (z = y; z <= yi; z++)
    {
        gotoxy(x, z);
        printf("%c", 179);
        gotoxy(xi, z);
        printf("%c", 179);
    }
    gotoxy(x, y);
    printf("%c", 218);
    gotoxy(xi, yi);
    printf("%c", 217);
    gotoxy(xi, y);
    printf("%c", 191);
    gotoxy(x, yi);
    printf("%c", 192);
}

/*Funcion que me permite poner el cargando en el centro de la pantalla*/
void centrartexto(char *texto, int y)
{
    int longitud = strlen(texto);
    gotoxy(40 - (longitud / 2), y);
    printf(texto);
}
