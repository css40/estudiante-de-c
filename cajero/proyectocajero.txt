enum TipoCaracter
{
    ES_LETRA,
    ES_NUMERO,
    NO_ES_LETRA_NI_NUMERO
};

enum TipoCaracter validarCaracter(char caracter)
{
    if (isalpha(caracter) != 0)
    {
        return ES_LETRA;
    }
    else if (isdigit(caracter) != 0)
    {
        return ES_NUMERO;
    }
    else
    {
        return NO_ES_LETRA_NI_NUMERO;
    }
}

        return;

    fflush(stdin);
    /* Limpia el búfer de entrada para evitar problemas */
    scanf("%8s", numero);

    /* Validar cada carácter del número de teléfono utilizando esLetra */
    for (i = 0; i < 8; i++)
    {
        if (esLetra(numero[i]))
        {
            cprintf("Numero de telefono no valido. Debe contener solo numeros.\n");textcolor(15);
            cprintf("Ingrese el numero telefonico (8 digitos): ");textcolor(15);
            fflush(stdin);
            /* Limpia el búfer de entrada nuevamente */
            scanf("%8s", numero);
            i = -1;
            /* Reiniciar el bucle para volver a validar desde el principio */
        }
    }

    cprintf("Ingrese la cantidad a recargar: ");textcolor(15);
    scanf("%f", &cantidad);

    if (cantidad < 0 || cantidad > saldo)
    {
        cprintf("Cantidad no valida. Debe ser mayor que 0 y menor o igual al saldo actual.\n");textcolor(15);
        getch();
        recarga();
        return;
        /* Salir de la función si la cantidad no es válida */
    }
    else
    {
        saldo -= cantidad;
        cprintf("Saldo actual: %.2f\n", saldo);textcolor(15);
        getch();
        recarga();
    }
    getch();
}

int validarCredenciales()
{
    
    char usuarioCorrecto[] = "u";       
    char contrasenaCorrecta[] = "c"; 
    char usuario[50];
    char contrasena[50];
    int intentos = 3;
    clrscr();

    do
    {
        printf("Ingrese su usuario: ");
        scanf("%s", usuario);
        printf("Ingrese su contraseña: ");
        scanf("%s", contrasena);
        clrscr();

        if (strcmp(usuario, usuarioCorrecto) == 0 && strcmp(contrasena, contrasenaCorrecta) == 0)
        {
            return 1;
        }
        else
        {
            intentos--;
            if (intentos > 0)
            {
                printf("Credenciales incorrectas. Tiene %d intentos restantes.\n", intentos);
            }
            else
            {
                printf("Ha agotado sus intentos. Saliendo del programa.\n");
                return 0;
            }
        }
    } while (intentos > 0);

    return 0; /*Agotó los intentos*/
}
